  <behavior>
    horizon_years = horizon_days/365 (default 1.0)

    entry_cost(venue, horizon_years) = (entry_slippage_bps/1e4)/horizon_years
    stressed_exit_cost(venue, hurdle, horizon_years) = (exit_slippage_bps/1e4)/0.30/horizon_years + hurdle·delay_days/365
      - lido 0.00099, curve 0.0050, aave 0.0005, paxg 0.0020 (±1e-5) at hurdle 0.0215, 1y

    venue_pd(venue, assessments) = 1 − (1 − pd_base)·(1 − max failure_excess over venue.exposures with an assessment)
      - no exposures / normal market → ≈ pd_base (within 0.0005)
      - USDC in distress snapshot raises aave_v3_usdc PD above pd_base

    score_venue(venue, quote, snapshot, assessments, lam, affinity=0.0, w_a=0.003, horizon_days=365) → ScoreBreakdown
      excess = apy − hurdle; EL = PD·LGD; loss_weighted = (1+λ)·EL;
      total = excess − loss_weighted − c_in − c_out_stress + w_a·affinity
      Demo snapshot, λ=1, affinity 0 (±0.0003):
      - lido_stake → +0.0055
      - curve_steth_eth → +0.0300
      - aave_v3_usdc → +0.0160
      - paxg_hold → −0.0061 (entry 3 bps adds −0.0003 vs spec table; update BRAIN_DESIGN row if it differs > 0.0001)
      Properties: higher λ never raises a score; positive affinity raises total by exactly w_a·a; breakdown terms sum to total.

    recovery_edge(price, assessment, quality, costs, hurdle, recovery_days=7) → float
      P_rec = p_normal + p_distress; V_fail = q·FV
      E = P_rec·(FV − p) − p_failure·(p − V_fail) − costs − hurdle·recovery_days/365
      - usdc_svb_2023 inputs, costs 0.005 → +0.031 ± 0.002
      - ust_2022_early inputs, costs 0.01 → −0.14 ± 0.003
  </behavior>
